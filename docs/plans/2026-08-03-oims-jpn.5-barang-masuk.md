## Context

Barang masuk = pencatatan bahan dari supplier (PRD §10). Efek:
1. Insert `barang_masuk` (header) + `barang_masuk_detail` (per bahan).
2. Untuk tiap detail: **append `mutasi_stok`** (tipe `masuk`, kuantitas positif) — JANGAN update `stok.kuantitas` langsung.
3. Update `bahan.hargaRataRata` (weighted average) untuk tiap bahan.
4. `stok.kuantitas` di-update lewat trigger DB (dari mutasi) — kalau trigger belum ada, update manual di action DALAM transaksi yang sama, TAPI turunannya tetap dari mutasi (bukan edit arbitrary).

Schema: `barangMasuk`, `barangMasukDetail` (sudah ada). Nomor dokumen `BM-YYYYMM-NNNN`.

---

## Invariant Wajib

| Aturan | Implementasi |
|---|---|
| Nomor dokumen auto | `BM-YYYYMM-NNNN`, generate di action, unique. Count barang_masuk bulan ini + 1. |
| Weighted average | `hargaBaru = (stokLama × hargaLama + qtyMasuk × hargaMasuk) / (stokLama + qtyMasuk)`. Update `bahan.hargaRataRata`. |
| Append mutasi | tiap detail → 1 baris `mutasi_stok` (tipe `masuk`, kuantitas = qty positif, barangMasukId = header id). |
| Atomik | Semua di `db.transaction()` — header + detail + mutasi + update harga sekaligus. Gagal 1 → rollback semua. |
| Audit | 1 entry CREATE untuk header (data_after = header + detail). |

## Task 1 — Nomor dokumen helper (`src/lib/document-number.ts`)

Reusable untuk BM/BK/PS. CREATE:
```typescript
import { db } from "@/db";
import { sql } from "drizzle-orm";

/**
 * Generate nomor dokumen: [TIPE]-YYYYMM-NNNN
 * Counter reset per bulan. Query count existing bulan ini + 1.
 * @param prefix "BM" | "BK" | "PS"
 * @param table nama tabel (untuk count)
 */
export async function generateDocNumber(
  prefix: string,
  tableName: "barang_masuk" | "barang_keluar" | "penyesuaian_stok"
): Promise<string> {
  const now = new Date();
  const yyyymm = `${now.getFullYear()}${String(now.getMonth() + 1).padStart(2, "0")}`;
  const like = `${prefix}-${yyyymm}-%`;
  const [{ count }] = await db.execute<{ count: number }>(
    sql`SELECT COUNT(*)::int as count FROM ${sql.identifier(tableName)} WHERE nomor_dokumen LIKE ${like}`
  );
  const nomor = String(count + 1).padStart(4, "0");
  return `${prefix}-${yyyymm}-${nomor}`;
}
```
> Race condition: dua transaksi bersamaan bisa dapat nomor sama → unique constraint DB reject → retry. Untuk MVP cukup; kalau sering tabrakan, tambah advisory lock. (ponytail: naive count, tambah lock kalau throughput tinggi)

```typescript
});

export type BarangMasukInput = z.infer<typeof barangMasukSchema>;
export type BarangMasukDetailInput = z.infer<typeof barangMasukDetailSchema>;
```

## Task 3 — Server Action createBarangMasuk (`src/services/barang-masuk.ts`)

**INTI. Semua dalam transaksi:**
```typescript
"use server";
import { db } from "@/db";
import { barangMasuk, barangMasukDetail, mutasiStok, bahan, stok, auditLog } from "@/db/schema";
import { eq } from "drizzle-orm";
import { requireRole } from "@/lib/auth";
import { generateDocNumber } from "@/lib/document-number";
import type { BarangMasukInput } from "@/lib/schemas/barang-masuk";

export async function createBarangMasuk(input: BarangMasukInput) {
  const user = await requireRole(["owner", "admin_gudang"]);

  return db.transaction(async (tx) => {
    const nomorDokumen = await generateDocNumber("BM", "barang_masuk");

    // 1. header
    const [header] = await tx.insert(barangMasuk).values({
      nomorDokumen,
      supplierId: input.supplierId || null,
      nomorInvoice: input.nomorInvoice || null,
      tanggal: new Date(input.tanggal),
      catatan: input.catatan || null,
      createdBy: user.id,
    }).returning();

    // 2. per detail: insert detail + mutasi + update harga rata2 + update stok cache
    for (const d of input.detail) {
      const subtotal = d.kuantitas * d.hargaSatuan;
      await tx.insert(barangMasukDetail).values({
        barangMasukId: header.id, bahanId: d.bahanId,
        kuantitas: String(d.kuantitas), hargaSatuan: String(d.hargaSatuan),
        subtotal: String(subtotal),
      });

      // append mutasi (append-only ledger)
      await tx.insert(mutasiStok).values({
        bahanId: d.bahanId, tipe: "masuk", kuantitas: String(d.kuantitas),
        barangMasukId: header.id, createdBy: user.id,
      });

      // weighted average
      const [b] = await tx.select().from(bahan).where(eq(bahan.id, d.bahanId)).limit(1);
      const [s] = await tx.select().from(stok).where(eq(stok.bahanId, d.bahanId)).limit(1);
      const stokLama = Number(s?.kuantitas ?? 0);
      const hargaLama = Number(b.hargaRataRata);
      const totalQty = stokLama + d.kuantitas;
      const hargaBaru = totalQty > 0
        ? (stokLama * hargaLama + d.kuantitas * d.hargaSatuan) / totalQty
        : d.hargaSatuan;
      await tx.update(bahan).set({ hargaRataRata: String(hargaBaru), updatedAt: new Date() })
        .where(eq(bahan.id, d.bahanId));

      // update stok cache (turunan dari mutasi — bukan edit arbitrary)
      await tx.update(stok).set({ kuantitas: String(totalQty), updatedAt: new Date() })
        .where(eq(stok.bahanId, d.bahanId));
    }

    // audit
    await tx.insert(auditLog).values({
      userId: user.id, aksi: "CREATE", tabel: "barang_masuk", recordId: header.id,
      dataBefore: null, dataAfter: JSON.stringify({ header, detail: input.detail }),
    });

    return { data: header };
  });
}
```
