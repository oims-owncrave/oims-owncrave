## Task 2 — Helper getPageTitle(pathname)

Di `sidebar/data/index.ts` (atau file util baru `nav-utils.ts`):
```ts
export function getPageTitle(pathname: string): string {
  for (const section of NAV_DATA) {
    for (const item of section.items) {
      if (item.url === pathname) return item.title;
      const sub = item.items.find((s) => s.url === pathname);
      if (sub) return sub.title;
    }
  }
  return "OIMS";
}
```
- Handle nested route (mis. `/inventory/barang-masuk/baru` → cari prefix match / fallback ke parent title + "Baru").

```tsx
"use client";
export function MobileHomeHeader({ userName, userRole }: { userName: string; userRole: string }) {
  const initials = userName.split(" ").filter(Boolean).slice(0, 2).map(w => w[0]?.toUpperCase()).join("");
  return (

      <div className="flex items-center gap-3">

          <p className="text-sm text-white/80">Selamat datang,</p>
          <h2 className="truncate text-lg font-semibold">{userName}</h2>
          <p className="text-xs capitalize text-white/70">{userRole}</p>
        </div>
      </div>
    </div>
  );
}
```
- Dark mode: gradient primary sudah kontras, teks putih OK di dark. Cek token `primary` ada di globals.css.
- Canonical classes: `size-11`, `pt-6`, `pb-5` — canonical. `bg-linear-to-r` (v4, WAJIB bukan gradient-to).

## Task 4 — MobilePageHeader (judul + back)

File BARU `header/mobile-page-header.tsx`:
```tsx
"use client";
import { useRouter } from "next/navigation";
export function MobilePageHeader({ title, showBack }: { title: string; showBack: boolean }) {
  const router = useRouter();
  return (
    <div className="min-[850px]:hidden flex items-center gap-2 border-b border-stroke bg-white px-4 py-3 dark:border-dark-3 dark:bg-gray-dark">
      {showBack && (
        <button onClick={() => router.back()} aria-label="Kembali" className="rounded-lg p-1.5 hover:bg-gray-2 dark:hover:bg-dark-2">
          <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
            <path d="M19 12H5M12 19l-7-7 7-7" />
          </svg>
        </button>
      )}
      <h1 className="truncate text-lg font-semibold text-dark dark:text-white">{title}</h1>
    </div>
  );
}
```
- **Rule showBack:** true kalau route punya segment lebih dalam dari list utama — mis. `/inventory/barang-masuk/baru`, `/inventory/barang-masuk/[id]`. False untuk list util (`/inventory/stok`, `/laporan/mutasi`). Heuristik: `showBack = pathname.split("/").length > 3` (mis. `/inventory/barang-masuk` = 3 segment → list, false; `/inventory/barang-masuk/baru` = 4 → true). Verifikasi tiap route saat coding.
