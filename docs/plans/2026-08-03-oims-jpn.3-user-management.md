## Task 2 — Helper getCurrentUser + requireRole

**File:** `src/lib/auth.ts` (CREATE)

```typescript
import { createClient } from "@/lib/supabase/server";
import { db } from "@/db";
import { users } from "@/db/schema";
import { eq } from "drizzle-orm";

export type CurrentUser = typeof users.$inferSelect;

export async function getCurrentUser(): Promise<CurrentUser | null> {
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return null;

  const [profile] = await db
    .select()
    .from(users)
    .where(eq(users.id, user.id))
    .limit(1);

  return profile ?? null;
}

export async function requireRole(
  allowedRoles: Array<typeof users.$inferSelect["role"]>
): Promise<CurrentUser> {
  const user = await getCurrentUser();
  if (!user) throw new Error("Unauthenticated");
  if (!user.isActive) throw new Error("Akun dinonaktifkan");
  if (!allowedRoles.includes(user.role)) throw new Error("Akses ditolak");
  return user;
}
```

**Verifikasi:** TypeScript strict, tidak ada `any`.

## Task 3 — Server Actions user

**File:** `src/services/user.ts` (CREATE)

Pattern identik master data: audit log tiap aksi, soft delete. Beda: create pakai Supabase Admin API.

```typescript
"use server";

import { createClient } from "@/lib/supabase/server";
import { createClient as createAdminClient } from "@supabase/supabase-js";
import { db } from "@/db";
import { users, auditLog } from "@/db/schema";
import { eq, isNull, and } from "drizzle-orm";
import { requireRole, getCurrentUser } from "@/lib/auth";
import type { UserInput } from "@/lib/schemas/user";

// Admin client — service_role key, server-only
function adminClient() {
  return createAdminClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.SUPABASE_SERVICE_ROLE_KEY!,
    { auth: { autoRefreshToken: false, persistSession: false } }
  );
}

async function writeAudit(
  aksi: string, recordId: string, before: unknown, after: unknown, userId: string | null
) {
  await db.insert(auditLog).values({
    userId: userId ?? undefined,
    aksi, tabel: "users", recordId,
    dataBefore: before ? JSON.stringify(before) : null,
    dataAfter: after ? JSON.stringify(after) : null,
  });
}

export async function listUsers() {
  await requireRole(["owner"]);
  return db.select().from(users).where(isNull(users.deletedAt ?? undefined));
  // catatan: users tidak punya deletedAt di schema — query tanpa filter deletedAt
}

export async function createUser(input: UserInput) {
  const currentUser = await requireRole(["owner"]);

  const email = `${input.username.toLowerCase()}@owncrave.local`;

  // 1. Buat auth.users via Admin API
  const admin = adminClient();
  const { data: authData, error: authError } = await admin.auth.admin.createUser({
    email,
    password: input.password,
    email_confirm: true,
    user_metadata: { username: input.username, full_name: input.displayName },
    app_metadata: { role: input.role, username: input.username },
  });
  if (authError || !authData.user) {
    return { error: authError?.message ?? "Gagal membuat akun auth" };
  }

  // 2. Insert public.users
  const [newUser] = await db.insert(users).values({
    id: authData.user.id,
    email,
    displayName: input.displayName,
    role: input.role,
    isActive: true,
  }).returning();

  await writeAudit("CREATE", newUser.id, null, newUser, currentUser.id);
  return { data: newUser };
}

export async function updateUser(
  id: string,
  input: Partial<Pick<UserInput, "displayName" | "role" | "isActive">>
) {
  const currentUser = await requireRole(["owner"]);

  const [before] = await db.select().from(users).where(eq(users.id, id)).limit(1);
  if (!before) return { error: "User tidak ditemukan" };

  // Prevent owner demoting themselves
  if (id === currentUser.id && input.role && input.role !== "owner") {
    return { error: "Tidak bisa mengubah role diri sendiri" };
  }

  const [updated] = await db.update(users)
    .set({ ...input, updatedAt: new Date() })
    .where(eq(users.id, id))
    .returning();

  // Sync role ke auth app_metadata jika role berubah
  if (input.role && input.role !== before.role) {
    const admin = adminClient();
    await admin.auth.admin.updateUserById(id, {
      app_metadata: { role: input.role },
    });
  }

  await writeAudit("UPDATE", id, before, updated, currentUser.id);
  return { data: updated };
}

export async function deactivateUser(id: string) {
  const currentUser = await requireRole(["owner"]);
  if (id === currentUser.id) return { error: "Tidak bisa menonaktifkan diri sendiri" };

  const [before] = await db.select().from(users).where(eq(users.id, id)).limit(1);
  if (!before) return { error: "User tidak ditemukan" };

  const [updated] = await db.update(users)
    .set({ isActive: false, updatedAt: new Date() })
    .where(eq(users.id, id))
    .returning();

  // Revoke Supabase session
  const admin = adminClient();
  await admin.auth.admin.signOut(id, "global");

  await writeAudit("DEACTIVATE", id, before, updated, currentUser.id);
  return { data: updated };
}
```

**Catatan penting:**
- `users` tabel tidak punya `deletedAt` — gunakan `isActive: false` sebagai soft-disable (bukan soft delete).
- `listUsers` — owner sees all. Tak ada deletedAt filter.
- `updateUser` — sync role ke `app_metadata` supaya token JWT ikut update (next login).
- Guard self-demotion + self-deactivation.

## Task 5 — TanStack Query Hook

**File:** `src/hooks/useUser.ts` (CREATE)

```typescript
"use client";

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { listUsers, createUser, updateUser, deactivateUser } from "@/services/user";
import { toast } from "sonner";

export function useUsers() {
  return useQuery({
    queryKey: ["users"],
    queryFn: () => listUsers(),
  });
}

export function useCreateUser() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: createUser,
    onSuccess: (res) => {
      if (res.error) { toast.error(res.error); return; }
      toast.success("User berhasil dibuat");
      qc.invalidateQueries({ queryKey: ["users"] });
    },
  });
}

export function useUpdateUser() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: ({ id, input }: { id: string; input: Parameters<typeof updateUser>[1] }) =>
      updateUser(id, input),
    onSuccess: (res) => {
      if (res.error) { toast.error(res.error); return; }
      toast.success("User diperbarui");
      qc.invalidateQueries({ queryKey: ["users"] });
    },
  });
}

export function useDeactivateUser() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: deactivateUser,
    onSuccess: (res) => {
      if (res.error) { toast.error(res.error); return; }
      toast.success("User dinonaktifkan");
      qc.invalidateQueries({ queryKey: ["users"] });
    },
  });
}
```

```typescript
import { Input } from "@/components/ui/Input";

import type { users } from "@/db/schema";

type User = typeof users.$inferSelect;

const ROLE_OPTIONS = [
  { value: "owner", label: "Owner" },
  { value: "admin_gudang", label: "Admin Gudang" },
  { value: "admin_produksi", label: "Admin Produksi" },
  { value: "keuangan", label: "Keuangan" },
  { value: "viewer", label: "Viewer" },
];

interface Props {
  open: boolean;
  onClose: () => void;
  editUser?: User | null;
}

export function UserFormModal({ open, onClose, editUser }: Props) {
  const isEdit = !!editUser;
  const create = useCreateUser();
  const update = useUpdateUser();

  const createForm = useForm<UserInput>({ resolver: zodResolver(userSchema) });
  const editForm = useForm<UserUpdateInput>({ resolver: zodResolver(userUpdateSchema) });

  useEffect(() => {
    if (editUser) {

        displayName: editUser.displayName,
        role: editUser.role,
        isActive: editUser.isActive,

    if (!editUser) return;
```
